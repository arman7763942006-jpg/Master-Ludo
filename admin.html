import random import sys class LudoGame: def __init__(self, players): self.players = players # List of player names self.board_size = 52 # Main track spaces # Har player ke tokens ki current positions (-1 matlab home/base par hain) self.tokens = {player: [ -1, -1, -1, -1 ] for player in players} # Safe zones ya starting positions self.start_positions = { self.players[0]: 0, self.players[1]: 13, self.players[2]: 26, self.players[3]: 39 if len(self.players) > 3 else 0 } def roll_dice(self): return random.randint(1, 6) def display_status(self): print("\n" + "="*40) print("CURRENT TOKEN POSITIONS:") for player, positions in self.tokens.items(): print(f"{player}: {positions}") print("="*40 + "\n") def play(self): print("Ludo Game Start Ho Chuka Hai!") print(f"Players: {', '.join(self.players)}") while True: for player in self.players: self.display_status() input(f"{player} ki baari hai. Dice roll karne ke liye Enter dabayein...") dice = self.roll_dice() print(f"🎲 {player} ne roll kiya aur aaya: {dice}") # Check agar sabhi tokens base par hain aur 6 nahi aaya all_at_home = all(pos == -1 for pos in self.tokens[player]) if all_at_home and dice != 6: print(f"Aapke sabhi tokens ghar me hain aur 6 nahi aaya. Baari agle player ko milti hai.\n") continue # Token select karne ke liye input lein valid_tokens = [] for i, pos in enumerate(self.tokens[player]): if pos == -1 and dice == 6: valid_tokens.append(i) elif pos != -1 and pos != 99: # 99 matlab win ho chuka token valid_tokens.append(i) if not valid_tokens: print(f"Chalane ke liye koi valid token nahi hai.\n") continue print(f"Aapke valid tokens index: {valid_tokens}") token_idx = -1 while token_idx not in valid_tokens: try: token_idx = int(input(f"Kaun sa token chalana hai ({'/'.join(map(str, valid_tokens))}): ")) except ValueError: print("Kripya sahi number dalein.") # Move logic current_pos = self.tokens[player][token_idx] if current_pos == -1 and dice == 6: # Token ko bahar nikalna (Home se start position par) start_pos = self.start_positions[player] self.tokens[player][token_idx] = start_pos print(f"{player} ka token 1 board par aa gaya!") else: # Normal move on board new_pos = current_pos + dice if new_pos >= self.board_size: new_pos = new_pos % self.board_size # Loop around board # Check for cutting opponent tokens self.check_collision(player, new_pos) self.tokens[player][token_idx] = new_pos print(f"{player} ka token position {new_pos} par pahunch gaya.") # Check for win condition if all(pos == 99 for pos in self.tokens[player]): print(f"\n🎉 Badhaai ho! {player} ne Ludo game jeet liya hai! 🎉") sys.exit() print("-" * 30) def check_collision(self, current_player, target_pos): # Dusre players ke tokens ko check karna agar us pos par hain toh unhe wapas ghar (-1) bhej dena for player, positions in self.tokens.items(): if player == current_player: continue for i, pos in enumerate(positions): if pos == target_pos: self.tokens[player][i] = -1 print(f"⚡ {current_player} ne {player} ke token ko kaat diya! Woh wapas ghar chala gaya.") if __name__ == "__main__": # Game setup (2 to 4 players) p_list = ["Player 1", "Player 2", "Player 3", "Player 4"] game = LudoGame(p_list) game.play()
